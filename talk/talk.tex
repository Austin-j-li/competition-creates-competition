\begin{frame}[label=F1b]{Does a stronger incumbent keep the challenger out?}
\framesubtitle{Earlier theory holds the challenger's information fixed}
\hypertarget{main:evidence}{}
\begin{itemize}\setlength{\itemsep}{8pt}
\item \textbf{Earlier theory} says yes
\item A stronger incumbent raises the price the challenger pays when it wins
\item So preparation pays less, and fewer challengers enter \graycite{Fishman 1988; Hirshleifer and Png 1989}
\item This answer holds the challenger's information fixed
\item Here the challenger sees the stock price before it decides whether to enter
\end{itemize}
\TakeawayWithNav[\ExhibitReadingGapRoomy]{\KeyIdea{The claim:} the stock price can flip the answer. A stronger incumbent can pull the challenger in.}

% ======================================================================
% Frame 4. This paper (punchline; label F2 kept from the earlier order)
% ======================================================================
\begin{frame}[label=F2]{This paper}
An investor who knows the challenger's value trades the target's stock before the challenger decides. The stock is worth what the sale pays. The model solves for trade and entry in equilibrium.
\vspace{2pt}
\begin{itemize}\setlength{\itemsep}{5pt}
\item \KeyIdea{Competition creates competition.} A stronger incumbent can make an uninformative stock price informative. Entry rises, although the challenger's expected auction profit before any news falls
\item \textbf{Why:} a strong incumbent can beat a low-value challenger, but cannot beat a high-value one. Therefore, the sale price depends more on the challenger's value. The investor trades on this, and good news in the price makes costly preparation pay
\item \textbf{The channel is information.} Hold fixed what the challenger learns from the price, and a stronger incumbent cannot raise entry
\end{itemize}
\Takeaway[0.6em]{A stronger incumbent does not always keep the challenger out. When the target's stock trades, entry depends on what the price reveals first.}
\end{frame}

\vspace{-2pt}
{\small
\begin{itemize}\setlength{\itemsep}{2pt}
\item \textbf{Sale rule:} a cash second-price auction with reserve price $p$, the minimum the seller accepts (not the stock price). The highest bid wins and pays the larger of $p$ and the other bid. Each bidder bids its value
\item \textbf{Incumbent:} value $R$, uniform on $[0,r]$. The strength $r$ is public; only the incumbent knows $R$
\item \textbf{Challenger:} type $\theta=H$ or $L$, equally likely, with value $h$ or $\ell$. It learns $\theta$ only if it prepares
\item $0<p<\ell<r<h$: the incumbent can beat a low-value challenger, but cannot beat a high-value one
\item \Cost{\textbf{Cost:} one challenger draws one cost: $c_L$ with probability $\rho$, otherwise $c_H$, independent of $\theta$}
\end{itemize}}

% ======================================================================
% Frame 6. Trading and the stock price (new; label F4b keeps later labels)
% ======================================================================
\begin{frame}[label=F4b]{Trading and the stock price}
\framesubtitle{Before entry: the challenger sees the price and its own cost, not the order flow, $\theta$ or $R$}
\hypertarget{main:eq}{}\hypertarget{main:orders}{}\hypertarget{main:laplace}{}\hypertarget{main:notation}{}
\begin{itemize}\setlength{\itemsep}{6pt}
\item \textbf{Investor:} knows $\theta$ and places an order $q$ in $[-1,1]$, at a trading cost of $k$ per unit. Its orders after $H$ and after $L$ are $(q_H,q_L)$
\item \textbf{Order flow:} $X=q+Z$, where the noise $Z$ is Laplace with scale $b>1$
\item \textbf{Price:} a share pays the sale proceeds. The model sets its payoff without a sale to zero. Market makers set $P=\mathbb{E}[\text{share payoff}\mid X]$, allowing for entry and the auction
\item \textbf{Belief:} $\mu=\Pr(\theta=H\mid P)$, the probability of high value after the price. For any orders, noise keeps $m\le\mu\le M$, where $m=1/(1+e^{2/b})$ and $M=1-m$
\item \textbf{Equilibrium:} each agent maximizes expected monetary payoff given the others. Prices equal expected payoffs; beliefs follow Bayes' rule. The investor may randomize its order (mixed orders) and deviate to any $q$ in $[-1,1]$
\end{itemize}
\NavGap
\PlaceNav{\hyperlink{app:eq}{\beamergotobutton{Equilibrium}}\NavSep\hyperlink{app:orders}{\beamergotobutton{Orders, units, costs}}\NavSep\hyperlink{app:laplace}{\beamergotobutton{Why Laplace noise}}\NavSep\hyperlink{app:notation}{\beamergotobutton{Notation}}}
\end{frame}

\begin{frame}[label=F5]{One auction, two claims, opposite responses}
\framesubtitle{Cash second-price auction with reserve $p$: who wins and who pays at incumbent value $R$}
\hypertarget{main:payoffs}{}\hypertarget{main:payrule}{}
\vspace{4pt}
\begin{center}\small
\begin{tabular}{@{}llll@{}}
\toprule
Incumbent value & High-value challenger & Low-value challenger & Gap in target proceeds\\
\midrule
$R\le\ell$ & wins, pays $\max\{p,R\}$ & wins, pays $\max\{p,R\}$ & $0$\\
$R>\ell$ & wins, pays $R$ & loses; incumbent pays $\ell$ & $R-\ell$\\
\bottomrule
\end{tabular}
\end{center}
\vspace{-2pt}
{\small As incumbent strength $r$ rises:\par}
\vspace{-8pt}
\begin{align*}
\Info{\Delta_T=t_H-t_L=\mathbb{E}[(R-\ell)_+]}\ &\Info{\uparrow} && \text{\Info{target-payoff spread = expected gap (last column)}}\\
g_H=\mathbb{E}[(h-\max\{p,R\})_+]\ &\downarrow && \text{high-value challenger's profit before preparation cost}\\
B_r(\mu)=g_L+\mu(g_H-g_L)\ &\downarrow && \text{challenger's expected gross profit at belief $\mu$}
\end{align*}
\vspace{-6pt}
\GrayLine{$t_\theta$ = expected target proceeds when a $\theta$-challenger enters. Replacing $h$ with $\ell$ in the profit formula gives $g_L$. Proposition 1: both signs hold weakly for any first-order strengthening, $R\sim F$ on $[0,\bar r]$, $\ell<\bar r<h$ (strictly in the uniform case, $\bar r=r$).}
\TakeawayWithNav{One shift, opposite signs: the challenger keeps less, and target proceeds depend more on its value.}
\PlaceNav{\hyperlink{app:payoffs}{\beamergotobutton{Closed forms}}\NavSep\hyperlink{app:payrule}{\beamergotobutton{Other payment rules}}}
\end{frame}

\Cost{\textbf{Low-cost floor}}\enspace $c_L<B_{r_1}(m)$: cheap preparation pays at every price, so entry is at least $\rho$\par

\begin{frame}[label=F7]{Trading pays only against a strong incumbent}
\framesubtitle{When does what the investor knows cover the trading cost $k$?}
\hypertarget{main:bound}{}
\vspace{4pt}
{\small \textbf{Residual advantage per unit, before the trading cost:}
\begin{align*}
\text{Buy after }H:&\quad \text{expected share payoff}-\text{price}\\
\text{Sell after }L:&\quad \text{price}-\text{expected share payoff}
\end{align*}\par}
\vspace{2pt}
{\centering\small
\begin{tabular}{@{}l>{\centering\arraybackslash}p{2.5cm}c>{\centering\arraybackslash}p{2.5cm}c>{\centering\arraybackslash}p{2.8cm}@{}}
\toprule
Residual advantage per unit $=$ & entry probability\newline ($\ge\rho$, low-cost floor) & $\times$ & \Info{target-payoff spread $\Delta_T$} & $\times$ & belief in other type\newline $1-\mu$ after $H$\newline $\mu$ after $L$\newline (both $\ge m$)\\
\bottomrule
\end{tabular}\par\vspace{3pt}
$\rho\,m\,\Info{\Delta_T}\ \le\ \text{advantage}\ \le\ \Info{\Delta_T}$\par}
{\small
\begin{itemize}\setlength{\itemsep}{2pt}
\item \textbf{Weak $r_0$:} advantage $\le\Delta_T(r_0)<k$, so every nonzero order loses in expectation: no trade, $(0,0)$
\item \textbf{Strong $r_1$:} each extra unit earns at least $(1-1/b)\,\rho m\Delta_T(r_1)>k$, so full orders $(1,-1)$
\end{itemize}}
\GrayLine{\Info{\textbf{Trading-cost window}} $\Delta_T(r_0)<k<(1-1/b)\,\rho m\Delta_T(r_1)$. The factor $1-1/b$ allows for the order's own price impact.}
\TakeawayWithNav[0.3em]{The same shift that lowers the challenger's profit switches informed trading on.}
\PlaceNav{\hyperlink{app:bound}{\beamergotobutton{Why full orders are optimal}}}
\end{frame}

\begin{frame}[label=F9]{Proposition 2: a stronger incumbent can raise entry}
\framesubtitle{Weak $r_0$, strong $r_1$, stronger still $r_2$; all other primitives equal}
\hypertarget{main:proof}{}\hypertarget{main:margins}{}\hypertarget{main:forces}{}
\begin{ResultBox}[Proposition 2, under the three conditions below]
\small
\PropItem{(i)}\textbf{Weak $r_0$:} no trade, $(q_H,q_L)=(0,0)$; the price is uninformative; entry $=\rho$

\begin{frame}[label=F15]{Three predictions}
\framesubtitle{Public sales: the incumbent is ready; the challenger can still choose whether to prepare}
\begin{itemize}\setlength{\itemsep}{8pt}
\item \textbf{A stronger incumbent can draw the challenger in} when the stock price comes before the entry decision. Without that price, a stronger incumbent cannot raise entry
\item \textbf{A challenger that prepares after good news wins more often.} In the benchmark, the high-value share among challengers that prepare equals $\mu$, the probability of high value implied by the price
\item \textbf{Beyond the ceiling, entry no longer responds to price.} Cheap entry and profitable trade must persist. The price stays informative, but even the best news cannot cover $c_H$
\end{itemize}
\medskip
\GrayLine{Outcome: the start of diligence. A valid test must use the price before that decision. Returns before a bid can also reflect an expected bid (Schwert 1996; Online Appendix D).}
\end{frame}

\begin{frame}[label=F16]{Conclusion}

\item \textbf{Why:} a stronger incumbent makes sale proceeds depend more on the challenger's value. The investor trades on this information. Good news then makes high-cost preparation worthwhile.

\begin{frame}[label=A7]{Equilibrium and the two comparisons}
\hypertarget{app:eq}{}
\begin{itemize}\setlength{\itemsep}{6pt}
\item \textbf{Equilibrium:} the investor may randomize its order. The price accounts for entry; the challenger updates its belief and chooses whether to prepare. Bidders bid their values. The investor may deviate to any $q\in[-1,1]$
\item Order flow $X=q+Z$, Laplace noise $Z$ with scale $b$; stock price $P(X)$ = expected target payoff given $X$
\item \textbf{Two comparisons kept apart:} to check one investor order, hold the price and entry rules fixed. To compare incumbent strengths, solve those rules again
\item \textbf{Uniqueness} covers investor orders and entry at prices that occur in equilibrium, under truthful bids
\end{itemize}

% ---------------------------------------------------------------- A10
\begin{frame}[label=A10]{Why the investor's order is unique}
\hypertarget{app:bound}{}
\begin{itemize}\setlength{\itemsep}{4pt}
\item \textbf{Weak:} gain per unit is at most $\Delta_T(r_0)<k$. Every nonzero order loses in expectation; the belief stays at $1/2$
\item \textbf{Strong:} let $s\in[0,1]$ be order size in the profitable direction. $\Pi_\theta(s)$ is expected gain per unit before cost; $U_\theta(s)$ is expected profit:
\end{itemize}
\vspace{-4pt}
\[
U_\theta(s)=s\,\Pi_\theta(s)-ks,\qquad U_\theta'(s)\ \ge\ (1-1/b)\,\rho m\,\Info{\Delta_T(r_1)}-k\ >\ 0\quad\text{on }[0,1]
\]
\vspace{-8pt}
\begin{itemize}\setlength{\itemsep}{4pt}
\item The bound holds for every order size under any candidate equilibrium with the low-cost entry floor, even when investor orders mix
\item \textbf{False good news:} an investor who knows value is low has a negative expected gain if it buys to mimic good news. It still pays $k$
\item Incumbent or target manipulation and toeholds: outside the model (A4)
\end{itemize}
\BackButton{main:bound}
\end{frame}

% ---------------------------------------------------------------- A11
\begin{frame}[label=A11]{Orders, units and the trading cost}
\hypertarget{app:orders}{}
\begin{itemize}\setlength{\itemsep}{6pt}
\item $q\in[-1,1]$ is a small order in target shares, measured in the same units as noise. It gives no control of the target. The investor starts with no position
\item $k$ is the cost per unit to execute or hold the position. The order also affects the price; that price effect is separate
\item \textbf{Why full orders are optimal:} gain per unit is at least $\rho m\Delta_T$. The noise bound limits the loss from price impact to a fraction $1/b$
\item The expected gain from each extra unit exceeds $k$ throughout $[0,1]$. Full orders are the unique best response; A10 gives the bound
\item Against the weak incumbent, $\Delta_T(r_0)<k$, so every nonzero order loses in expectation
\end{itemize}
\BackButton{main:orders}
\end{frame}

\begin{frame}[label=A13]{The price reveals the market's belief}
\hypertarget{app:suff}{}
\begin{itemize}\setlength{\itemsep}{3pt}
\item Even when the investor randomizes its order, Laplace noise bounds the belief from flow (likelihood ratio within $e^{\pm2/b}$):
\end{itemize}
\vspace{-6pt}
\begin{gather*}
\mu_X(x)=\Pr(H\mid X=x)\in[m,M]\\
P=t_0+e(P)[t_L-t_0+\Delta_T\mu_X]\\
\mu_X=\frac{P-t_0-e(P)(t_L-t_0)}{e(P)\Delta_T}
\end{gather*}
\vspace{-6pt}
\begin{itemize}\setlength{\itemsep}{3pt}
\item $t_0$ = expected target proceeds without a challenger; $e(P)$ = preparation probability after price $P$
\item $e(P)\ge\rho>0$ and $\Delta_T>0$, so the denominator is positive. The price therefore reveals $\mu_X$
\item This also holds when some prices have positive probability or entry jumps at a threshold
\end{itemize}
\Takeaway{The challenger can recover the market's probability of high value from the price alone.}

\begin{frame}[label=A15]{Why the theorem assumes low-cost preparation always pays}
\hypertarget{app:floor}{}
\begin{itemize}\setlength{\itemsep}{6pt}
\item The \Cost{low-cost floor} keeps entry $\ge\rho$ after every price. The price reveals the market's belief; the investor's gain per unit is at least $\rho m\Delta_T$
\item If low-cost preparation does not pay after bad news, entry can stop there. Different flows can then receive the same price
\item If low-cost preparation does not pay at the prior, no preparation and no trade can be an equilibrium. This statement assumes that no other source links target value to challenger value
\item At the benchmark, $B_{r_1}(m)-c_L=1.37$: low-cost preparation pays even after the worst news % source: base_margin_low_cost
\item With the price hidden, entry is 0.250 % source: base_hidden_entry_weak, base_hidden_entry_strong

% ---------------------------------------------------------------- A16
\begin{frame}[label=A16]{Proof logic in four steps}
\hypertarget{app:proof}{}
\setbeamertemplate{enumerate items}{\textbf{\insertenumlabel.}}
\begin{enumerate}\setlength{\itemsep}{5pt}
\item \textbf{Weak economy:} no nonzero order covers $k$ in expectation; no trade; entry $\rho$
\item \textbf{Strong economy:} the global bound (A10) forces full orders
\item Under full orders, the belief needed for costly preparation lies in $(1/2,M)$. Let $\alpha_\theta$ be the chance of news above that threshold given type $\theta$. Then $\alpha_H>\alpha_L$: extra entry favors high-value challengers
\item \textbf{Information:} the strong price can reproduce the weak signal by discarding information. A type-independent rule cannot do the reverse. This is strict Blackwell dominance
\end{enumerate}
\medskip
\textbf{Part (iii):} with the floor and profitable trading still in place at $r_2$, $B_{r_2}(M)<c_H$ removes expensive entry.
\Takeaway{Step 2 rules out other trading outcomes, even when the investor may randomize its order.}
\BackButton{main:proof}
\end{frame}

\begin{frame}[label=A20]{How entry and ownership are computed}
\hypertarget{app:entry}{}
{\small At $r_1$, full orders $(1,-1)$ and Laplace noise give:\par}

\begin{gather*}

\end{itemize}}
