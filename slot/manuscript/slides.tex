\begin{frame}[t,label=S1]{Does a stronger incumbent always keep the rival out?}
\framesubtitle{Why the answer matters}
\hypertarget{main:s1}{}
\vspace{2.2em}
\begin{columns}[T,totalwidth=\linewidth]
\column{0.31\linewidth}
\ColHead{Seller}A public sale must find a rival bidder. A stronger incumbent can help find one, or end the search.
\column{0.31\linewidth}
\ColHead{Rival}Preparation costs money before a bid is possible. The stock price can show whether the cost may pay.
\column{0.31\linewidth}

% ---------------------------------------------------------------- Frame 2
\begin{frame}[t,label=S2]{The received answer: strength cannot raise entry}
\framesubtitle{When what the rival can learn stays fixed}
\hypertarget{main:s2}{}
\vspace{0.6em}
\begin{itemize}\setlength{\itemsep}{8pt}
\item The rival pays to learn its value before it can bid.
\item A stronger incumbent weakly cuts the rival's expected auction profit at a fixed belief.
\item Some preparation costs then fail to pay. Entry cannot rise.
\item Fishman (1988): a chosen preemptive bid. Here: a public strength shift. The value itself stays private.
\end{itemize}
\TakeawayEnd{This holds when pre-preparation information stays fixed.}
\FootLine{Fishman (1988); Hirshleifer and Png (1989). Auction entry: Samuelson (1985); McAfee and McMillan (1987); Levin and Smith (1994); Gentry and Stroup (2019).}
\end{frame}

\begin{frame}[t,label=S4]{This paper: a stronger incumbent can recruit the rival}
\hypertarget{main:s4}{}
\Body
\LargeLine{A stronger incumbent weakly cuts the rival's profit at each belief. It can also widen the shareholder spread. Informed trade can then pay, and the price can recruit the rival.}
\vspace{0.4em}
\begin{enumerate}\setlength{\itemsep}{4pt}
\item \textbf{Non-monotone entry:} entry can rise from the weak to the strong incumbent, then return to its floor. Each case has a unique outcome (Proposition 2).
\item \textbf{Value split:} strength weakly cuts rival profit and weakly widens the shareholder spread (Proposition 1).
\item \textbf{Information control:} with pre-preparation information fixed, a stronger incumbent cannot raise entry.
\end{enumerate}
\vspace{0.5em}

\begin{frame}[t,label=S5]{Players and the sale rule}
\hypertarget{main:s5}{}
\ModelStrip{1}
\Body
{\raggedright A listed target has standalone value $0$. Values and costs are per share. All agents are risk neutral.\par}
\begin{DefList}
\DefRow{Incumbent}{\textbf{incumbent bidder, already prepared:} it knows its value $R$ and faces no further preparation cost. $R\sim F$; $F$ is public, continuous on $[0,\bar r]$, and others do not observe the draw before bids. \textbf{Stronger} means a replacement $\widetilde F$ with $\widetilde F(y)\le F(y)$ for each $y$.}
\DefRow{Rival}{\textbf{potential rival bidder:} sees the price, then decides whether to prepare. \textbf{Synergy class:} $j=H$ (good synergy) or $j=L$ (poor synergy), with prior $\pi=\Pr(j=H)\in(0,1)$. Its value $\theta$ has law $G_j$ in class $j$; each $G_j$ has a finite mean.}
\DefRow{Class ranking}{$G_H(y)\le G_L(y)$ for each $y$ --- class $H$ draws weakly higher values. Also $\theta\ge p$ almost surely under each $G_j$, where $p>0$ is the auction reserve below. The sale and the rival's identity are public; its synergy class is not public.}
\DefRow{Preparation cost $C$}{$C=0$ with probability $\rho\in(0,1)$; else $C=c_H>0$, the high cost ($H$ in $c_H$ means high, not class $H$). Pay $C$ to learn $\theta$ and to bid; a rival who does not pay cannot bid. ($\rho$ is the entry floor on frame 9.)}
\DefRow{Sale}{the seller commits before trade to a cash second-price auction with reserve $p>0$. Sale proceeds go to target shareholders.}
\end{DefList}
\vfill
\LightRule
{\footnotesize\raggedright The pair $(j,\theta)$, the value $R$, the cost $C$, and the trading-slot draws are independent. Neither bidder trades target shares. The next frame adds the trading slot: informed trader, liquidity trader, competitive market makers.\par}
\end{frame}

\vspace{-0.5em}
\LightRule
\begin{itemize}\setlength{\itemsep}{1pt}
\item One trading slot comes before preparation. Its outcome is $X\in\{-1,0,1\}$: $-1$ stock sale, $0$ no trade, $1$ stock purchase.
\item With probability $\lambda\in(0,1)$ --- the informed presence --- the slot goes to an outside informed trader. It observes the synergy class $j$ and chooses an order $q\in\{-1,0,1\}$; each nonzero order costs $k>0$.
\item With probability $1-\lambda$, a liquidity trader fills the slot and submits $-1$, $0$, $1$ with probabilities $\beta$, $1-2\beta$, $\beta$, where $0<\beta\le 1/3$.

\end{itemize}

\hypertarget{main:s7}{}

\Body
{\footnotesize\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{4.9cm}>{\raggedright\arraybackslash}p{4.3cm}>{\raggedright\arraybackslash}p{5.1cm}@{}}
\toprule
Player & Observes & Chooses\\
\midrule

\vspace{0.5em}
\begin{itemize}\setlength{\itemsep}{3pt}
\item \textbf{Order belief:} $\mu(x)=\Pr(j=H\mid X=x)$ --- the belief of an observer who sees the realised outcome $X=x$.
\item \textbf{Price belief:} $\bar\mu(\bar P)=\Pr(j=H\mid P(X)=\bar P)$ --- the rival's belief. One price pools all outcomes with that price.
\end{itemize}

% ---------------------------------------------------------------- Frame 8
\begin{frame}[t,label=S8]{Payoffs}
\hypertarget{main:s8}{}
\ModelStrip{5}
\Body
\begin{center}\SplitLine{6.2}{incumbent's value $R$}\end{center}
\vspace{-0.5em}
\begin{itemize}\setlength{\itemsep}{2pt}
\item Fix a prepared rival with value $\theta\ge p$, and average over $R$. Shareholders receive the auction payment $\max\{p,\min(R,\theta)\}$; its mean is $t(\theta)$. With $(z)_+=\max\{z,0\}$, the rival's gross profit is $g(\theta)=\mathbb E_R[(\theta-\max\{p,R\})_+]$, before cost $C$. In expectation, $t(\theta)+g(\theta)=\theta$.
\item $\mathbb E_j$ denotes expectation under $G_j$. Class means are $t_j=\mathbb E_j[t(\theta)]$ and $g_j=\mathbb E_j[g(\theta)]$. With no rival entry, the incumbent alone pays $p$ if $R\ge p$; expected proceeds are $t_0=p[1-F(p)]$.
\item \textbf{Shareholder spread:} $\Delta_T=t_H-t_L$ --- extra expected target proceeds in class $H$ versus class $L$ if the rival prepares. \textbf{Rival gross profit} at belief $\mu$ is $B(\mu)=g_L+\mu(g_H-g_L)$, the $\mu$-weighted mean of class gross profits, before $C$. Net gain from preparation: $B(\text{price belief})-C$.
\item The informed trader with order $q$ earns $q[V_T-P(q)]-k|q|$, where $P(q)$ is the price after order $q$.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- Frame 9
% A label column: the four parts of the equilibrium as rows under the solution concept.
\begin{frame}[t,label=S9]{Equilibrium with competitive prices}
\hypertarget{main:s9}{}
\ModelStrip{6}
\Body
{\raggedright Solution concept: perfect Bayesian equilibrium with break-even prices from competitive market makers.\par}
\vspace{0.15em}
\begin{DefList}[3.0cm]
\DefRow{Orders}{let $e(x)$ be the probability that the rival prepares after realised outcome $X=x$, through the price $P(x)$. An informed trader who sees class $j$ gets expected net payoff $U_j(q)=q[t_0+e(q)(t_j-t_0)-P(q)]-k|q|$ from chosen order $q$; the term $t_0+e(q)(t_j-t_0)$ is expected target proceeds in class $j$. For each $j$, the informed trader puts positive probability only on orders $q$ that maximize $U_j(q)$, with the schedules $P(\cdot)$ and $e(\cdot)$ at their equilibrium values.}
\DefRow{Competitive prices}{$P(x)=\mathbb{E}[V_T\mid X=x]=t_0+e(x)[t_L-t_0+\Delta_T\,\mu(x)]$. With entry, expected proceeds are $t_L+\mu(x)\Delta_T$; without entry they are $t_0$. Zero-cost preparation below gives the entry floor $e(x)\ge\rho$.}
\DefRow{Beliefs}{the order belief $\mu(x)=\Pr(j=H\mid X=x)$ follows Bayes' rule. The rival's price belief $\bar\mu(\bar P)=\Pr(j=H\mid P(X)=\bar P)$ pools all outcomes with that price. It can differ from $\mu(x)$ for any one pooled outcome. Liquidity trade gives each outcome positive probability in each class, so Bayes' rule pins down order beliefs.}
\DefRow{Preparation and bids}{a rival with $C=0$ prepares at each price. With $C=c_H$, it prepares if and only if $B(\bar\mu)\ge c_H$ at its price belief. A prepared rival bids $\theta$; the incumbent bids $R$ subject to the reserve.}
\end{DefList}
\end{frame}

\begin{frame}[t,label=S10]{Strength never raises rival profit and never narrows the spread}
\hypertarget{main:s10}{}
\Body\TightMath
\begin{Statement}{Proposition 1}
\raggedright
Let $F$ and $\widetilde F$ be continuous incumbent value distributions,
with $\widetilde F(y)\le F(y)$ for every $y$.
\Gloss{The new incumbent draws high values more often.}
Hold the reserve and the finite-mean class laws fixed.\par
\vspace{0.1em}
\begin{minipage}[t]{0.46\linewidth}\raggedright
(a) For $j\in\{H,L\}$ and every $\mu\in[0,1]$,
\[
g_j(\widetilde F)\le g_j(F),\qquad
B_{\widetilde F}(\mu)\le B_F(\mu).
\]
Here $B_F(\mu)=g_L(F)+\mu[g_H(F)-g_L(F)]$.
\end{minipage}\hfill
\begin{minipage}[t]{0.51\linewidth}\raggedright
(b) Under the class ranking,
\[
G_H(y)\le G_L(y)\quad\text{for every }y,\qquad
G_H(p^-)=G_L(p^-)=0,
\]
the shareholder spread satisfies $\Delta_T(\widetilde F)\ge\Delta_T(F)$.
\Gloss{Good-synergy values are higher, and all meet the reserve.}
\end{minipage}\par
\vspace{0.1em}
Since both classes meet the reserve almost surely,
$\Delta_T(F)+[g_H(F)-g_L(F)]
=\mathbb E_H[\theta]-\mathbb E_L[\theta]$.
The right side does not depend on $F$.
\end{Statement}
\par\centerline{\scalebox{0.6}{\GapLine{5.4}{8.2}}}\par
\Intuition{A stronger incumbent can move part of the rival's gain from good synergy to the target's shareholders.}
\end{frame}

% ---------------------------------------------------------------- Frame 11
% Rows: the definitions full width; the band and the price-reveals-belief part side by side; the

\begin{frame}[t,label=S11]{Beliefs stay in a band, and the price reveals the belief}
\hypertarget{main:s11}{}
\Body\TightMath
\begin{Statement}{Belief bounds}
\raggedright
Define\quad $b=(1-\lambda)\beta,\qquad u=\dfrac{\lambda}{b},\qquad
m(\pi)=\dfrac{\pi}{\pi+(1-\pi)(1+u)},\qquad
M(\pi)=\dfrac{\pi(1+u)}{\pi(1+u)+1-\pi}.$\par
\begin{minipage}[t]{0.46\linewidth}\raggedright
For every pure or mixed order strategy and every $x\in\{-1,0,1\}$,
\[
0<\beta\le\frac13
\quad\Longrightarrow\quad
m(\pi)\le\mu(x)\le M(\pi).
\]
\Gloss{With $\beta\le1/3$, liquidity trade covers every outcome, including no trade.}
The price belief $\bar\mu$ obeys the same bounds.
\end{minipage}\hfill
\begin{minipage}[t]{0.51\linewidth}\raggedright
Suppose $\Delta_T>0$, $g_H>g_L$ and $\rho\in(0,1)$.
Under Bayes' rule, optimal preparation, competitive prices and truthful bids,
\[
\begin{aligned}
&P(x)=P(x')\ \Longrightarrow\ \mu(x)=\mu(x'),\\
&\Pr(H\mid P(X)=P(x))=\mu(x).
\end{aligned}
\]
\end{minipage}\par
\begin{minipage}[t]{0.46\linewidth}\raggedright
\textbf{Preparation rule.} Define the threshold $\tau=(c_H-g_L)/(g_H-g_L)$.
\Gloss{At the belief $\tau$, rival gross profit equals the high cost.}
Under the default preparation rule,
\Gloss{A rival at the threshold prepares.}
\end{minipage}\hfill
\begin{minipage}[t]{0.51\linewidth}\raggedright
\[
\begin{aligned}
&C=c_H:\quad\text{prepare exactly when }\mu(x)\ge\tau,\\
&e(x)=\begin{cases}\rho,&\mu(x)<\tau,\\1,&\mu(x)\ge\tau.\end{cases}
\end{aligned}
\]
\end{minipage}
\end{Statement}
\Intuition{A liquidity trader can place any order, so an order can move the belief but never settles the synergy class.}
\PlaceNav{\hyperlink{app:b2}{\beamergotobutton{B2}}}
\end{frame}

% ---------------------------------------------------------------- Frame 12
\begin{frame}[t,label=S12]{A correct order earns between a fixed share and all of the spread}
\hypertarget{main:s12}{}
\Body\TightMath
\begin{Statement}{Trading bounds}
\raggedright
Let $\Delta_T\ge0$ and $k>0$.
Take any order profile with Bayes beliefs, competitive prices, truthful bids,
and $e(x)\in[\rho,1]$.
\Gloss{The bounds hold for every trading plan, optimal or not.}
The gross gains from the correct orders are
\[
K_H=e(1)\Delta_T[1-\mu(1)],\qquad
K_L=e(-1)\Delta_T\mu(-1).
\]
Define $\kappa(\pi)=\min\{m(\pi),1-M(\pi)\}$.
\Gloss{The value $\kappa(\pi)$ is the least doubt any order leaves.}
For either class $j\in\{H,L\}$,
\[
\rho\kappa(\pi)\Delta_T\le K_j\le\Delta_T.
\]
A wrong-signed order earns at most $-k<0$; abstention earns zero.
The informed trader who sees class $H$ buys if $K_H>k$ and abstains if $K_H<k$.
The informed trader who sees class $L$ sells if $K_L>k$ and abstains if $K_L<k$.
At $K_j=k$, the correct order and abstention are both optimal, and any mixture of the two is optimal.
\end{Statement}
\Intuition{Gross gain scales with the shareholder spread and with the doubt an order leaves. Net gain subtracts $k$, so doubt alone does not pay.}
\PlaceNav{\hyperlink{app:b2}{\beamergotobutton{B2}}}
\end{frame}

The outcome is unique in each of the three economies,

\end{itemize}

\begin{itemize}\setlength{\itemsep}{4pt}
\item \textbf{Lemma A.3} (informed trading): the gain before order cost $k$ from a correct purchase under good synergy is $K_H=e(1)\,\Delta_T\,[1-\mu(1)]$; from a correct sale under poor synergy, $K_L=e(-1)\,\Delta_T\,\mu(-1)$. Under every order plan, $\rho\,\Delta_T[1-M(\pi)]\le K_H\le\Delta_T$ and $\rho\,\Delta_T\,m(\pi)\le K_L\le\Delta_T$. With $\kappa(\pi)=\min\{m(\pi),1-M(\pi)\}$, each gain lies between $\rho\,\kappa(\pi)\,\Delta_T$ and $\Delta_T$; the purchase bound uses $1-M(\pi)$ and the sale bound uses $m(\pi)$. A wrong-signed order earns at most $-k$. Good synergy buys when $K_H>k$ and stays out when $K_H<k$; poor synergy compares $K_L$ with $k$ in the same way.
\item Four-step proof: 1.~Weak: each gain is at most $\Delta_T$, below $k$, so no informed order; the belief stays at the prior; entry stays at $\rho$. 2.~Strong: each gain is at least $\rho\,\kappa(\pi)\,\Delta_T$, above $k$, so full orders hold under every candidate; beliefs are $(m(\pi),\pi,M(\pi))$. 3.~A purchase reaches $M(\pi)$, rival gross profit there strictly exceeds the high cost, so entry after a purchase is $1$; after other orders entry stays at $\rho$. 4.~Very strong: full orders stay, but rival gross profit at $M(\pi)$ falls short of the high cost, so entry returns to $\rho$ after each order.
\end{itemize}

\vspace{1.0em}
\begin{itemize}\setlength{\itemsep}{12pt}
\item The informed trader is an outsider with sector expertise. It holds no net position in the target before trade. It has no control rights, and it cannot acquire the target.

\item Win chance: $W_j(F)=\mathbb{E}_j[F(\theta)\mathbf 1\{\theta\ge p\}]$. A good-synergy rival outbids $F$ when $W_H(F)=1$.

\vspace{0.6em}
\begin{itemize}\setlength{\itemsep}{10pt}
\item With class gap $D(y)=G_L(y)-G_H(y)$: $\Delta_T(F)=pF(p)D(p^-)+\int_p^\infty[1-F(y)]D(y)\,dy$ and $g_j(F)=\int_p^\infty[1-G_j(y)]F(y)\,dy$.
\item Profit falls strictly exactly when $\int_p^\infty[1-G_j(y)][F(y)-\tilde F(y)]\,dy>0$. Spread rises strictly exactly when $\int_p^\infty[F(y)-\tilde F(y)]D(y)\,dy>0$.
\item Without the class ranking, the reserve term $pF(p)D(p^-)$ can reverse the spread sign. With no class atom at the reserve, the spread weakly rises under every strengthening exactly when class $H$ dominates above the reserve and the classes have equal admissibility, $G_H(p^-)=G_L(p^-)$.
\end{itemize}

\item (E6) Truthful bids: each prepared bidder with value at or above reserve $p$ bids its value.
\item Tie rules, default pair: an indifferent informed trader may mix over its best orders, and an indifferent rival at cost $c_H$ prepares. Other rules: an indifferent informed trader abstains, or an indifferent rival at cost $c_H$ prepares with any probability.
\end{itemize}

\item The price gap $P(1)-P(-1)$ splits into a synergy part, $\rho\Delta_T[M(\pi)-m(\pi)]$, and an entry part. The entry part is positive exactly under recruitment, and it sits wholly in the purchase impact.
\end{itemize}
\vspace{0.1em}
\begin{enumerate}\setlength{\itemsep}{0.5pt}
\item Entry can rise with incumbent strength only when the price lands before preparation starts. Hide the price, and entry can only fall.
\item Past the entry ceiling, orders stop predicting entry. Orders keep predicting the winner when $W_H>W_L$.
\item At prior $1/2$, order counts cannot separate slot types. Prices can.
\end{enumerate}}
\vspace{0.05em}

\begin{frame}[t,label=B13b]{Predictions in full}
\BackupId{B13}
\small
\vspace{0.4em}
\begin{itemize}\setlength{\itemsep}{5pt}
\item Mean order under full two-sided trade is $\lambda(2\pi-1)$. Counts name informed presence only when the prior is known and differs from $1/2$.
\item At prior $1/2$, sale impact plus volume name presence when entry sits at its floor after a sale and after no order, with payoff scale $\rho\Delta_T$ known.
\item For a prepared rival, the winner gap after purchase against sale equals $[M(\pi)-m(\pi)](W_H-W_L)$, with win chance $W_j$ in class $j$. The prediction is strict only when $W_H>W_L$.
\item A report that reveals preparation alone carries a positive mean price change exactly when preparation stayed uncertain after some order. Both reports together average to zero.
\item At one middle strength, sale processes can form two clusters: uninformative prices with mean order zero, or informative prices with mean order $\lambda/2$.
\item Count preparation starts, not public offers alone. Date incumbent strength from public records before the preparation decision.
\end{itemize}

\begin{frame}[t,label=B13c]{Predictions in full}
\framesubtitle{Weighted price impact and the winner · Proposition A.10}
\BackupId{B13}
\small
\vspace{0.6em}
\begin{itemize}\setlength{\itemsep}{6pt}
\item Let order frequency be $\bar f(x)=\Pr(X=x)$, purchase impact $I_+=P(1)-P(0)$, and sale impact $I_-=P(0)-P(-1)$. \textbf{Proposition A.10:} under full two-sided trade, class $H$ buys and class $L$ sells with probability one, and prior $\pi<{}$threshold $\tau$, the weighted difference $\bar f(1)I_+-\bar f(-1)I_-$ is positive exactly when entry after a purchase exceeds the floor, $e(1)>\rho$. At prior $\pi=1/2$ the frequencies agree, and the raw impacts $I_+$ and $I_-$ compare directly. The entry part of the price gap $P(1)-P(-1)$ sits in purchase impact $I_+$.
\item Under the same full two-sided trade, where entry $e(x)=\rho$ after each order, the order $X$ does not predict entry. It still predicts the winner exactly when the win chance $W_H$ exceeds $W_L$, with $W_j=\mathbb{E}_j[F(\theta)\mathbf{1}\{\theta\ge p\}]$ for a prepared rival of class $j$; the indicator $\mathbf{1}\{\theta\ge p\}$ is one when the value meets the reserve. For a prepared rival, the win-probability gap after a purchase against a sale is $[M(\pi)-m(\pi)](W_H-W_L)$.
\end{itemize}

\begin{frame}[t,label=B15]{Continuous order sizes}
\hypertarget{app:b15}{}\BackupId{B15}
\small
\vspace{0.4em}
\begin{itemize}\setlength{\itemsep}{4pt}
\item The slot set-up uses unit orders $\{-1,0,1\}$, so every order is on path and no off-path belief enters
\item A menu of continuous order sizes needs a convention for the sizes that no synergy class uses, as in Edmans, Goldstein and Jiang 2015
\item Additive liquidity flow also keeps every size on path
\item With Laplace noise of scale $\xi$ and full unit orders, the top belief equals the slot value $M(\pi)$ when $1+u=\mathrm{e}^{2/\xi}$
\item The two set-ups then share the belief bounds. They differ in the probability that each synergy class reaches the top belief
\end{itemize}
\Takeaway{Unit orders keep prices on path at every order.}
