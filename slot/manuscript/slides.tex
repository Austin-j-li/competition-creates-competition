\begin{frame}[t,label=S1]{Does a stronger bidder in a merger auction always deter others from entering?}
\framesubtitle{Why the answer matters}
\vspace{2.2em}
\begin{columns}[T,totalwidth=\linewidth]
\column{0.315\linewidth}
\RuledCol{\textbf{Sellers must find the second buyer.} A public sale tries to discover one more buyer. Does a visible strong buyer help the search, or end the search?}
\column{0.315\linewidth}
\RuledCol{\textbf{Buyers pay before they know.} Diligence costs money before any offer exists. Does a strong rival kill the case for diligence, or make the market signal worth more?}
\column{0.315\linewidth}
\RuledCol{\textbf{Prices move before bids move.} Early price action may cause the second bid. The price may not only forecast the bid.}
\end{columns}
\vfill
\FootLine{One more bidder carries first-order value in Bulow and Klemperer (1996). This paper holds the sale rule fixed and studies entry into the sale.}

\framesubtitle{Deterrence at fixed information}
\hypertarget{main:s2}{}
\vspace{1.4em}
\begin{itemize}\setlength{\itemsep}{10pt}

% ---------------------------------------------------------------- Slide 8
\begin{frame}[t,label=S8]{The auction splits each value between shareholders and the buyer}
\framesubtitle{Payoff objects · Paper §2.4 and §3.1}
\vspace{0.6em}
{\raggedright Take a prepared second buyer with value $\theta\ge p$. It pays $\max\{p,R\}$ when it wins. The first buyer pays $\theta$ when it wins.\par}
\[
t(\theta)=\max\{p,\min(R,\theta)\},\qquad g(\theta)=(\theta-\max\{p,R\})_+,\qquad t(\theta)+g(\theta)=\theta
\]
\begin{itemize}\setlength{\itemsep}{3pt}
\item Class means: $t_j=\mathbb{E}_j[t(\theta)]$ and $g_j=\mathbb{E}_j[g(\theta)]$ for $j\in\{H,L\}$. Without entry, $t_0=p[1-F(p)]$.
\item Target-payoff spread: $\Delta_T=t_H-t_L$. Read it as: how much more shareholders get with a good fit than with a poor fit.
\item At belief $\mu=\Pr(H)$, expected buyer profit is $B(\mu)=g_L+\mu(g_H-g_L)$.
\item Win chance: $W_j(F)=\mathbb{E}_j[F(\theta)\mathbf 1\{\theta\ge p\}]$. Class $H$ outbids $F$ when $W_H(F)=1$.
\end{itemize}
\TakeawayEnd{The traded claim and the entrant claim sum to the same value. Strength moves the split.}
\vspace{0.2em}
\end{frame}

% ---------------------------------------------------------------- Slide 9
\begin{frame}[t,label=S9]{Equilibrium: orders, prices, beliefs, preparation, bids}
\framesubtitle{Definition · Paper §2.4}
\hypertarget{main:s9}{}
{\small
\begin{enumerate}\setlength{\itemsep}{0pt}
\item \textbf{Orders.} One distribution over $\{-1,0,1\}$ for each class. Mixing and wrong-signed orders are allowed.
\item \textbf{Beliefs and prices.} $\mu(x)=\Pr(H\mid X=x)$ by Bayes' rule. Every order can come from each class, so no off-path belief arises. Prices are competitive:
\vspace{-0.7em}\[P(x)=t_0+e(x)\,\bigl[t_L-t_0+\Delta_T\,\mu(x)\bigr]\]\vspace{-1.25em}
\item \textbf{Preparation.} $e(x)$ is the chance the second buyer prepares after price $P(x)$. A zero bill always prepares. A bill $c_H$ prepares exactly at beliefs with $B(\mu)\ge c_H$.
\item \textbf{Investor choice.} Class $j$ picks $q$ to maximise $U_j(q)=q\,\bigl[t_0+e(q)(t_j-t_0)-P(q)\bigr]-k|q|$ at the candidate prices and entry rule.
\item \textbf{Bids.} Each prepared bidder with value at or above $p$ bids its value.
\end{enumerate}}
\SmallLine{Default tie rules: an indifferent investor may mix over best orders; an indifferent buyer at bill $c_H$ prepares. Proposition 2 holds under all four tie-rule pairs.}
\vspace{0.1em}
{\color{cLine}\rule{\linewidth}{0.4pt}}\par\vspace{0.15em}
{\footnotesize\raggedright Entry $\mathsf{E}$ = chance the second buyer prepares. Ownership $\mathsf{O}_H$ = chance a good-fit second buyer wins. Takeover share = chance the winner is a good fit, given the second buyer wins.\par}
\vspace{0.15em}
{\footnotesize\raggedright Strengthen $F$. Solve the equilibrium again in full. Nothing carries across economies except the primitives.\par}
\vfill
\PlaceNav{\hyperlink{app:b3}{\beamergotobutton{Backup B3}}}
\end{frame}

% ---------------------------------------------------------------- Slide 11
\begin{frame}[t,label=S11]{Strength takes from the buyer and gives to the signal}
\framesubtitle{Proposition 1 · Paper §3.1 · General result}
\hypertarget{main:s11}{}
\vspace{0.9em}
{\color{cInk}\rule{\linewidth}{0.4pt}}\par\vspace{0.35em}
\begin{enumerate}\setlength{\itemsep}{3pt}
\item At each fixed belief, a stronger first buyer weakly lowers expected buyer profit $B(\mu)$.
\item Under the class ranking, a stronger first buyer weakly raises the shareholder spread $\Delta_T$.
\end{enumerate}
\vspace{0.1em}{\color{cInk}\rule{\linewidth}{0.4pt}}\par
\[
\Delta_T(F)+g_H(F)-g_L(F)=\mathbb{E}_H[\max\{\theta,p\}]-\mathbb{E}_L[\max\{\theta,p\}]
\]
{\raggedright The right side does not depend on $F$. Strength leaves the fit gap the same size. Strength moves the gap between pockets.\par}
\TakeawayEnd{Deterrence and informativeness come from the same shift. One force creates the other.}
\FootLine{Strict signs hold under the overlap conditions in Proposition 1. Detail: Backup B5.}
\PlaceNav{\hyperlink{app:b5}{\beamergotobutton{Backup B5}}}
\end{frame}

% ---------------------------------------------------------------- Slide 12
\begin{frame}[t,label=S12]{One order can point. It cannot prove.}
\framesubtitle{Belief band and threshold · Paper §3.2 and §3.4 · General}
\vspace{-1.1em}
% split across two lines: the formula, then its scope
\begin{gather*}
m(\pi)=\frac{\pi}{\pi+(1-\pi)(1+u)}\ \le\ \mu\ \le\ \frac{\pi(1+u)}{\pi(1+u)+1-\pi}=M(\pi)\\
\text{after every order, under every strategy}
\end{gather*}
\vspace{-1.5em}
\begin{center}
\begin{tikzpicture}[x=9cm,y=0.8cm]
  \fill[cLine!55] (0.28,-0.11) rectangle (0.72,0.11);
  \draw[cMuted,line width=0.5pt] (0,0) -- (1,0);
  \foreach \x/\l in {0.28/{m(\pi)},0.5/{\pi},0.72/{M(\pi)}} {\draw[cInk,line width=0.8pt] (\x,-0.17) -- (\x,0.17); \node[below=1pt,font=\small] at (\x,-0.17) {$\l$};}
\end{tikzpicture}
\end{center}
\vspace{-0.9em}
{\small
\begin{itemize}\setlength{\itemsep}{0pt}
\item Price reads belief off. With entry at least $\rho$ and $\Delta_T>0$, the price rises strictly in $\mu$. The second buyer recovers $\mu$ from $P$.
\item Threshold $\tau=(c_H-g_L)/(g_H-g_L)$, when $g_H>g_L$. A bill $c_H$ pays exactly when $\mu\ge\tau$.
\item Exclusion bound: once $B(M(\pi))<c_H$, no order recruits a buyer at bill $c_H$, in equilibrium or out of equilibrium.
\item Two-numbers rule: when $\Delta_T>0$ and $g_H>g_L$, orders, beliefs, and preparation depend on the auction only through $\Delta_T$ and $\tau$.
\end{itemize}}
\TakeawayEnd{The best signal in this market stops at $M(\pi)$. Recruitment lives or dies at that cap.}
\FootLine{The two-numbers rule holds with all other primitives and tie rules fixed. Price levels also need $t_0$ and $t_L$. Ownership also needs $W_H$.}
\vspace{0.2em}
\end{frame}

\vspace{-0.4em}

K_H=e(1)\,\Delta_T\,[1-\mu(1)],\qquad K_L=e(-1)\,\Delta_T\,\mu(-1)

\vspace{-1.0em}
\[
\rho\,\kappa(\pi)\,\Delta_T\ \le\ K_H,\,K_L\ \le\ \Delta_T,\qquad \kappa(\pi)=\min\{m(\pi),\,1-M(\pi)\}

\vspace{-0.8em}

\column{0.56\linewidth}

\begin{itemize}\setlength{\itemsep}{1.5pt}

\item Rational pricing charges for expected entry. The band leaves residual doubt. The zero-bill floor $\rho$ turns doubt into a floor on gain.
\item A wrong-signed order never pays. Each class trades exactly when its correct-order gain beats $k$.
\end{itemize}}
\column{0.42\linewidth}
\begin{center}

% ---------------------------------------------------------------- Slide 14
\begin{frame}[t,label=S14]{A stronger first buyer can bring the second buyer in}
\framesubtitle{Proposition 2 · Paper §4.1 · General result}
\hypertarget{main:s14}{}
\vspace{-0.7em}
{\footnotesize\renewcommand{\arraystretch}{1.0}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.4cm}>{\raggedright\arraybackslash}p{2.5cm}>{\raggedright\arraybackslash}p{4.0cm}>{\raggedright\arraybackslash}p{3.7cm}@{}}
\toprule
 & Weak $F_0$ & Strong $F_1$ & Very strong $F_2$\\
\midrule
Fund orders & None & Buys on good fit; sells on poor fit & Buys on good fit; sells on poor fit\\
Beliefs after sale, no order, purchase & $\pi,\ \pi,\ \pi$ & $m(\pi),\ \pi,\ M(\pi)$ & $m(\pi),\ \pi,\ M(\pi)$\\
Price & One price; no news & $P(-1)<P(0)<P(1)$ & Three prices; still news\\
Entry & $\rho$ & $\boldsymbol{\rho+(1-\rho)(b+\pi\lambda)>\rho}$ & Back to $\rho$\\
\bottomrule
\end{tabular}}
\vspace{0.25em}
{\footnotesize
\begin{itemize}\setlength{\itemsep}{0pt}
\item (H1) $B_0(\pi)<c_H<B_1(M(\pi))$ — profit at the prior falls short of the bill at $F_0$; profit at top news covers the bill at $F_1$.
\item (H2) $\Delta_0<k<\rho\kappa(\pi)\Delta_1$ — trade cannot pay at $F_0$ and must pay at $F_1$.
\item (H3) $B_2(M(\pi))<c_H$ — even top news cannot pay the bill at $F_2$.
\end{itemize}
\vspace{0.05em}
{\raggedright Classes meet the ranking. $F_1$ first-order strengthens $F_0$. $F_2$ first-order strengthens $F_1$. All other primitives stay fixed.\par}
\vspace{0.05em}
{\raggedright Each outcome is unique under all four tie-rule pairs. Conditions (H1)--(H3) hold together on an open set.\par}
\vspace{0.05em}
{\raggedright Good-fit ownership rises exactly when $e_HW_H(F_1)>\rho W_H(F_0)$, where $e_H=\rho+(1-\rho)(b+\lambda)$. Ownership rises for sure when $W_H(F_1)=1$.\par}}
\TakeawayEnd{Entry rises, then returns. The price stays live after entry dies.}
\PlaceNav{\hyperlink{app:b6}{\beamergotobutton{Backup B6}}}
\end{frame}

\begin{frame}[label=B3]{Equilibrium in full: six conditions and tie rules}
\hypertarget{app:b3}{}
{\small
\begin{itemize}\setlength{\itemsep}{3pt}
\item \textbf{Tuple:} order laws for each class, a price for each order in $\{-1,0,1\}$, beliefs at each price, the entry rule, the bids
\item \textbf{Prices:} $P(X)=\mathbb{E}[V_T\mid X]$ at each order, with entry at that price inside $V_T$
\item \textbf{Beliefs:} Bayes rule from the order laws at each price. Every order is on path, so no off-path belief enters
\item \textbf{Entry:} the zero cost always prepares; cost $c_H$ prepares exactly at prices with $B_r(\mu)\ge c_H$
\item \textbf{Investor:} the order law puts weight only on best replies at the stated price and entry rules, net of cost $k$
\item \textbf{Bids:} each bidder bids its value in the second-price auction
\item \textbf{Tie rules:} a tied investor may mix; a tied cost $c_H$ type prepares. Proposition 2 holds under all four tie rules
\end{itemize}}
\vspace{0.4em}
{\small\raggedright Proposition 2 holds under all four tie-rule pairs. Propositions 3 and 4 use the default tie rules.\par}

\begin{itemize}\setlength{\itemsep}{10pt}
