\begin{frame}[t]{Players and the sale rule}
\ModelStrip{1}
\vspace{0em}
{\small
\begin{itemize}\setlength{\itemsep}{0pt}
\item A listed target has standalone value $0$. Values and costs are per share. All agents are risk neutral.
\item \textbf{First buyer --- already prepared:} it knows its value $R$ and faces no preparation choice. $R\sim F$; $F$ is public, continuous on $[0,\bar r]$, and others do not observe the draw before bids. \textbf{Stronger} means a replacement $\widetilde F$ with $\widetilde F(y)\le F(y)$ for each $y$.
\item \textbf{Second buyer --- synergy class:} $j=H$ (good synergy) or $j=L$ (poor synergy), with prior $\pi=\Pr(j=H)\in(0,1)$. Its value $\theta$ has law $G_j$ in class $j$; each $G_j$ has a finite mean.
\item \textbf{Class ranking:} $G_H(y)\le G_L(y)$ for each $y$ --- class $H$ draws weakly higher values. Also $\theta\ge p$ almost surely under each $G_j$, where $p>0$ is the auction reserve below. The sale and the second buyer's identity are public; its synergy class is not public.
\item \textbf{Preparation cost $C$:} $C=0$ with probability $\rho\in(0,1)$; else $C=c_H>0$, the high cost ($H$ in $c_H$ means high, not class $H$). Pay $C$ to learn $\theta$ and to bid; a second buyer who does not pay cannot bid. ($\rho$ is the entry floor on frame 9.)
\item \textbf{Sale:} the seller commits before trade to a cash second-price auction with reserve $p>0$. Sale proceeds go to target shareholders.
\item The pair $(j,\theta)$, the value $R$, the cost $C$, and the trading-slot draws are independent. Neither buyer trades target shares. The next frame adds the trading slot: informed investor, liquidity trader, market makers.
\end{itemize}}
\end{frame}

\begin{frame}[t]{Strength can raise entry, then return it to its floor}
\vspace{-1.3em}
{\footnotesize
\begin{block}{\ResultTag{Proposition 2}}\TightMath
\begin{columns}[T,totalwidth=\linewidth]
\column{0.43\linewidth}
Classes have finite means and satisfy $G_H\le G_L$, $G_H(p^-)=G_L(p^-)=0$. Steps $F_2\le F_1\le F_0$: weak, strong, very strong.
Fix the class laws and $p>0$; $\pi,\rho,\lambda\in(0,1)$; $\beta\in(0,1/2)$ (nonzero-order bounds only); and $c_H,k>0$.
Write $\Delta_i=\Delta_T(F_i)$ and $B_i=B_{F_i}$.
Suppose
\[
\begin{array}{ll}
\mathrm{(H1)}&B_0(\pi)<c_H<B_1(M(\pi)),\\
\mathrm{(H2)}&\Delta_0<k<\rho\kappa(\pi)\Delta_1,\\
\mathrm{(H3)}&B_2(M(\pi))<c_H.
\end{array}
\]
\Gloss{(H1) Prior misses $c_H$ at $F_0$; a purchase clears it at $F_1$.}
\Gloss{(H2) Order cost stops trade at $F_0$, not at $F_1$. (H3) At $F_2$, even a purchase cannot pay $c_H$.}
\column{0.55\linewidth}
\begin{center}\setlength{\tabcolsep}{2pt}
\begin{tabular}{@{}lll@{}}
\hline
Orders $(H,L)$ & Beliefs $(-1,0,1)$ & Entry $(-1,0,1)$\\
\hline
$F_0$ weak:\ $(0,0)$ & $(\pi,\pi,\pi)$ & $(\rho,\rho,\rho)$\\
$F_1$ strong:\ $(1,-1)$ & $(m(\pi),\pi,M(\pi))$ & $(\rho,\rho,1)$\\
$F_2$ very strong:\ $(1,-1)$ & $(m(\pi),\pi,M(\pi))$ & $(\rho,\rho,\rho)$\\
\hline
\end{tabular}
\end{center}
Let $\mathsf E_i$ denote total entry at $F_i$.
Then
\[
\mathsf E_0=\mathsf E_2=\rho
<\mathsf E_1=\rho+(1-\rho)(b+\pi\lambda).
\]
The outcome is unique in each of the three economies,
under all four pairs of tie rules.
\end{columns}
\end{block}}
\Intuition{A wider shareholder spread pays the informed investor to trade, and a purchase recruits. At $F_2$, profit at the top belief stays below $c_H$.}
\SmallLine{Proof idea: the frame 11--12 bounds hold for each candidate outcome; strict gaps then fix orders, beliefs, and entry. Steps: \hyperlink{app:b2}{B2}. Open set: frame 14. Exact form: \hyperlink{app:b1}{B1}.}
\end{frame}
